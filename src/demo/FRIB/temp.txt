using LinearAlgebra
using Random
using Distributions


function create_beam_from_envelope_matrix(envelope_matrix::Matrix{Float64}; centroid::Vector{Float64}, 
                                        nparticles::Int, energy::Float64=1e9, charge::Float64=-1.0, mass::Float64=931494320.0, seed::Int=42)
    # Validate input
    if size(envelope_matrix) != (6, 6)
        error("Envelope matrix must be 6x6")
    end
    
    if length(centroid) != 6
        error("Centroid must be a 6-element vector")
    end
    
    # Set random seed for reproducibility
    Random.seed!(seed)
    
    # Ensure the envelope matrix is symmetric
    envelope_matrix_symm = (envelope_matrix + envelope_matrix')/2
    
    # Decompose the envelope matrix to get the transformation
    # We'll use the Cholesky decomposition if possible
    L = try
        cholesky(envelope_matrix_symm).L
    catch
        # Fallback to eigendecomposition if Cholesky fails
        eigen_decomp = eigen(envelope_matrix_symm)
        values = eigen_decomp.values
        vectors = eigen_decomp.vectors
        
        # Replace any negative eigenvalues with small positive values
        values = max.(values, 1e-20)
        
        # Reconstruct L
        sqrt_values = sqrt.(values)
        L = vectors * Diagonal(sqrt_values)
    end
    
    # Generate standard normal distribution
    dist = Normal(0, 1)
    particles_raw = rand(dist, nparticles, 6)
    
    # Apply the transformation matrix to get correlated distribution
    particles_correlated = particles_raw * transpose(L)
    
    # Shift the distribution to be centered at the specified centroid
    particles_final = particles_correlated .+ reshape(centroid, 1, 6)
    
    # Create a Beam object
    beam = Beam(r=particles_final, np= nparticles, energy = energy, charge=charge, mass=mass)
    
    return beam
end

function verify_beam_parameters(beam::Beam, expected_centroid::Vector{Float64}, expected_envelope::Matrix{Float64})
    # Calculate the actual centroid
    actual_centroid = vec(mean(beam.r, dims=1))
    
    # Calculate the actual envelope matrix
    centered_coords = beam.r .- reshape(actual_centroid, 1, 6)
    actual_envelope = transpose(centered_coords) * centered_coords / size(beam.r, 1)
    
    # Calculate differences
    centroid_diff = actual_centroid - expected_centroid
    envelope_diff = actual_envelope - expected_envelope
    
    return centroid_diff, envelope_diff
end
